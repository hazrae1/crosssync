cmake_minimum_required(VERSION 3.16)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(CrossSync VERSION 1.1.0 LANGUAGES CXX)

add_library(crosssync_core src/profile.cpp)
target_include_directories(crosssync_core PUBLIC include)
target_compile_features(crosssync_core PUBLIC cxx_std_17)
set_target_properties(crosssync_core PROPERTIES CXX_EXTENSIONS OFF)

add_executable(crosssync src/main.cpp)
find_package(Threads REQUIRED)
target_link_libraries(crosssync PRIVATE crosssync_core Threads::Threads)
if(MSVC)
    target_compile_options(crosssync_core PRIVATE /W4 /permissive-)
    target_compile_options(crosssync PRIVATE /W4 /permissive-)
else()
    target_compile_options(crosssync_core PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(crosssync PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(crosssync_tests tests/profile_tests.cpp)
    target_link_libraries(crosssync_tests PRIVATE crosssync_core)
    add_test(NAME profile_validation COMMAND crosssync_tests)
    add_test(NAME cli_export_preview COMMAND crosssync --dry-run --fast --no-color)
    set_tests_properties(cli_export_preview PROPERTIES PASS_REGULAR_EXPRESSION "Configuration session complete")
    add_test(NAME cli_cs_specific_profile COMMAND crosssync --input ${PROJECT_SOURCE_DIR}/examples/cs-only.cfg --dry-run --fast --no-color)
    set_tests_properties(cli_cs_specific_profile PROPERTIES PASS_REGULAR_EXPRESSION "4 preserved")
    add_test(NAME cli_rejects_unknown_option COMMAND crosssync --unknown)
    set_tests_properties(cli_rejects_unknown_option PROPERTIES WILL_FAIL TRUE)
endif()

install(TARGETS crosssync RUNTIME DESTINATION bin)
